	Label.TextSize = 12
	Label.Font = Enum.Font.GothamMedium
	Label.TextXAlignment = Enum.TextXAlignment.Left
	Label.Parent = Holder

	Value.Name = "SliderValue"
	Value.BackgroundTransparency = 1
	Value.Position = UDim2.new(1, -65, 0, 7)

	Value.Text = tostring(Default)
	Value.TextColor3 = Accent
	Value.TextSize = 11
	Value.Font = Enum.Font.GothamBold
	Value.TextXAlignment = Enum.TextXAlignment.Right
	Value.Parent = Holder

	local Bar = Instance.new("Frame")
	Bar.Position = UDim2.fromOffset(12, 37)
	Bar.Size = UDim2.new(1, -24, 0, 7)
	Bar.BackgroundColor3 = Color3.fromRGB(50, 50, 60)
	Bar.BorderSizePixel = 0
	Bar.Parent = Holder

	local BarCorner = Instance.new("UICorner")
	BarCorner.CornerRadius = UDim.new(1, 0)
	BarCorner.Parent = Bar

	local Fill = Instance.new("Frame")
	Fill.Size = UDim2.fromScale(
		(Default - Min) / (Max - Min),
		1
	)
	Fill.BackgroundColor3 = Accent
	Fill.BorderSizePixel = 0
	Fill.Parent = Bar

	local FillCorner = Instance.new("UICorner")
	FillCorner.CornerRadius = UDim.new(1, 0)
	FillCorner.Parent = Fill

	local Sliding = false

	local function SetValueFromX(X)
		local Percent = math.clamp(
			(X - Bar.AbsolutePosition.X) / Bar.AbsoluteSize.X,
			0,
			1
		)

		local NewValue = math.floor(
			Min + ((Max - Min) * Percent) + 0.5
		)

		Value.Text = tostring(NewValue)
		Fill.Size = UDim2.fromScale(Percent, 1)

		Callback(NewValue)
	end

	Bar.InputBegan:Connect(function(Input)
		if Input.UserInputType == Enum.UserInputType.MouseButton1 then
			Sliding = true
			SetValueFromX(Input.Position.X)
		end
	end)

	UserInputService.InputChanged:Connect(function(Input)
		if Sliding and Input.UserInputType == Enum.UserInputType.MouseMovement then
			SetValueFromX(Input.Position.X)
		end
	end)

	UserInputService.InputEnded:Connect(function(Input)
		if Input.UserInputType == Enum.UserInputType.MouseButton1 then
			Sliding = false
		end
	end)

	Callback(Default)

	return Holder
end

--==============================================================
-- VISIBILITY CHECK
--==============================================================

local function IsPlayerVisible(TargetPlayer)
	local TargetCharacter = TargetPlayer.Character

	if not TargetCharacter then
		return false
	end

	local Head = TargetCharacter:FindFirstChild("Head")

	if not Head then
		return false
	end

	local CurrentCamera = workspace.CurrentCamera

	if not CurrentCamera then
		return false
	end

	local Origin = CurrentCamera.CFrame.Position
	local Direction = Head.Position - Origin

	local RaycastParamsObject = RaycastParams.new()

	RaycastParamsObject.FilterType = Enum.RaycastFilterType.Exclude
	RaycastParamsObject.FilterDescendantsInstances = {
		Character
	}

	RaycastParamsObject.IgnoreWater = true

	local Result = workspace:Raycast(
		Origin,
		Direction,
		RaycastParamsObject
	)

	if not Result then
		return true
	end

	return Result.Instance:IsDescendantOf(TargetCharacter)
end

--==============================================================
-- TEAM CHECK
--==============================================================

local function IsTeammate(TargetPlayer)
	if not Config.TeamCheck then
		return false
	end

	-- Do not treat every player as a teammate
	-- when the game does not use Roblox Teams.
	if Player.Team == nil then
		return false
	end

	return TargetPlayer.Team == Player.Team
end

local function GetESPColor(TargetPlayer)
	if Config.VisualCheck and not IsPlayerVisible(TargetPlayer) then
		return Color3.fromRGB(255, 50, 50)
	end

	return Accent
end

local function CreateESP(TargetPlayer)
	if TargetPlayer == Player then
		return
	end

	local TargetCharacter = TargetPlayer.Character

	if not TargetCharacter then
		return
	end

	local Existing = TargetCharacter:FindFirstChild("PlesneviyESP")

	if Existing then
		Existing:Destroy()
	end

	local Highlight = Instance.new("Highlight")
	Highlight.Name = "PlesneviyESP"
	Highlight.Adornee = TargetCharacter
	Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	Highlight.FillTransparency = 0.65
	Highlight.OutlineTransparency = 0
	Highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
	Highlight.FillColor = GetESPColor(TargetPlayer)
	Highlight.Parent = TargetCharacter
end

local function RemoveESP(TargetPlayer)
	local TargetCharacter = TargetPlayer.Character

	if not TargetCharacter then
		return
	end

	local Existing = TargetCharacter:FindFirstChild("PlesneviyESP")

	if Existing then
		Existing:Destroy()
	end
end

local function UpdateESP()
	for _, TargetPlayer in ipairs(Players:GetPlayers()) do
		if TargetPlayer ~= Player then
			if Config.PlayerESP then
				CreateESP(TargetPlayer)
			else
				RemoveESP(TargetPlayer)
			end
		end
	end
end

--==============================================================
-- AIMBOT TARGET
--==============================================================

local function GetClosestTarget()
	local ClosestPlayer = nil
	local ClosestDistance = Config.FOV

	local ViewportSize = Camera.ViewportSize
	local ScreenCenter = Vector2.new(
		ViewportSize.X / 2,
		ViewportSize.Y / 2
	)

	for _, TargetPlayer in ipairs(Players:GetPlayers()) do

		if TargetPlayer ~= Player then

			local TargetCharacter = TargetPlayer.Character
			local TargetHumanoid = TargetCharacter
				and TargetCharacter:FindFirstChildOfClass("Humanoid")

			local TargetHead = TargetCharacter
				and TargetCharacter:FindFirstChild("Head")

			if TargetHumanoid
				and TargetHumanoid.Health > 0
				and TargetHead then

				-- TEAM CHECK
				if IsTeammate(TargetPlayer) then
					continue
				end

				-- VISUAL CHECK
				if Config.VisualCheck
					and not IsPlayerVisible(TargetPlayer) then
					continue
				end

				local ScreenPosition, OnScreen =
					Camera:WorldToViewportPoint(TargetHead.Position)

				if OnScreen then
					local Distance = (
						Vector2.new(
							ScreenPosition.X,
							ScreenPosition.Y
						) - ScreenCenter
					).Magnitude

					if Distance < ClosestDistance then
						ClosestDistance = Distance
						ClosestPlayer = TargetPlayer
					end
				end
			end
		end
	end

	return ClosestPlayer
end

	FlyVelocity = Instance.new("BodyVelocity")
	FlyVelocity.Name = "PlesneviyFlyVelocity"
	FlyVelocity.MaxForce = Vector3.new(
		math.huge,
		math.huge,
		math.huge
	)
	FlyVelocity.Velocity = Vector3.zero
	FlyVelocity.Parent = Root
end

local function StopFly()
	if FlyVelocity then
		FlyVelocity:Destroy()
		FlyVelocity = nil
	end
end

--==============================================================
-- NPC FREEZE
--==============================================================

local FrozenNPCs = {}

local function UpdateNPCFreeze()
	if not Config.FreezeNPC then
		for NPC, Data in pairs(FrozenNPCs) do
			if NPC and NPC.Parent then
				local NPCHumanoid = NPC:FindFirstChildOfClass("Humanoid")

				if NPCHumanoid then
					NPCHumanoid.WalkSpeed = Data.WalkSpeed
					NPCHumanoid.JumpPower = Data.JumpPower
					NPCHumanoid.AutoRotate = Data.AutoRotate
				end
			end
		end

		table.clear(FrozenNPCs)
		return
	end

	for _, Object in ipairs(workspace:GetChildren()) do
		if Object:IsA("Model")
			and Object ~= Character
			and not Players:GetPlayerFromCharacter(Object) then

			local NPCHumanoid = Object:FindFirstChildOfClass("Humanoid")

			if NPCHumanoid then
				if not FrozenNPCs[Object] then
					FrozenNPCs[Object] = {
						WalkSpeed = NPCHumanoid.WalkSpeed,
						JumpPower = NPCHumanoid.JumpPower,
						AutoRotate = NPCHumanoid.AutoRotate
					}
				end

				NPCHumanoid.WalkSpeed = 0
				NPCHumanoid.JumpPower = 0
				NPCHumanoid.AutoRotate = false
			end
		end
	end
end

local function GetFlyDirection()
	local Direction = Vector3.zero

	if UserInputService:IsKeyDown(Enum.KeyCode.W) then
		Direction += Camera.CFrame.LookVector
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.S) then
		Direction -= Camera.CFrame.LookVector
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.A) then
		Direction -= Camera.CFrame.RightVector
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.D) then
		Direction += Camera.CFrame.RightVector
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
		Direction += Vector3.yAxis
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
		Direction -= Vector3.yAxis
	end

	if Direction.Magnitude > 0 then
		return Direction.Unit
	end

	return Vector3.zero
end

-- RENDER LOOP
--==============================================================

RunService.RenderStepped:Connect(function()
	Camera = workspace.CurrentCamera

	-- FOV
	if Config.Aimbot then
		FOVCircle.Visible = true

		local Viewport = Camera.ViewportSize

		FOVCircle.Position = UDim2.fromOffset(
			Viewport.X / 2,
			Viewport.Y / 2
		)

		local Diameter = Config.FOV * 2

		FOVCircle.Size = UDim2.fromOffset(
			Diameter,
			Diameter
		)

		FOVStroke.Color = Accent
	else
		FOVCircle.Visible = false
	end

	-- WALK SPEED
	if Config.WalkSpeedEnabled and Humanoid then
		Humanoid.WalkSpeed = Config.WalkSpeed
	end

	-- FLY
	if Config.Fly then
		if not FlyVelocity then
			StartFly()
		end

		if FlyVelocity then
			FlyVelocity.Velocity =
				GetFlyDirection() * Config.FlySpeed
		end
	else
		if FlyVelocity then
			StopFly()
		end
	end

	-- FULL BRIGHT
	UpdateFullBright()

	-- NPC FREEZE
	UpdateNPCFreeze()

	-- ESP
	if Config.PlayerESP then
		UpdateESP()
	else
		for _, TargetPlayer in ipairs(Players:GetPlayers()) do
			if TargetPlayer ~= Player then
				RemoveESP(TargetPlayer)
			end
		end
	end

	-- AIMBOT
	if Config.Aimbot and HoldingAim then
		local TargetPlayer = GetClosestTarget()

		if TargetPlayer
			and TargetPlayer.Character then

			local TargetHead =
				TargetPlayer.Character:FindFirstChild("Head")

			if TargetHead then
				local Smooth =
					math.clamp(
						Config.AimbotSmoothness / 100,
						0.01,
						1
					)

				local Alpha =
					math.clamp(
						1 - Smooth,
						0.01,
						0.99
					)

				local TargetCFrame =
					CFrame.lookAt(
						Camera.CFrame.Position,
						TargetHead.Position
					)

				Camera.CFrame =
					Camera.CFrame:Lerp(
						TargetCFrame,
						Alpha
					)
			end
		end
	end
end)
